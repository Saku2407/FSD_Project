# Inside the simulation() view:
student = {
    'name': data['name'],
    'course': data['course'],
    'marks': data['marks'],
    'skills': data['skills'],
    'status': 'PASS' if data['marks'] >= 40 else 'FAIL'
}
context['result'] = student
return render(request, 'views_template/simulation.html', context)
